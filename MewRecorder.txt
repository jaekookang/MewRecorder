       function [success, msg] = initialize(app)
            clc;

                return;
            end

                    % but libraries exist under MewRecoder
                    wrapperPath = fullfile(ctfroot, 'MewRecorder', relWrapper);
                else
                    wrapperPath = fullfile('C:\Users','Boram Kim','GitHub','MewRecorder',relWrapper);
                end
        
                % === Normalize safely (handle spaces) ===
                wrapperPath = char(java.io.File(wrapperPath).getCanonicalPath());  % keep as char, not string
        
                if exist(wrapperPath, 'dir')
                    if ~contains(path, wrapperPath)
                        addpath(wrapperPath);
                        app.logMessage(['[MewRecorder] Added NatNet wrapper path: ', wrapperPath]);
                    end
                else
                    app.logMessage(sprintf('NatNet wrapper path not found: %s', wrapperPath), 'error');
                end
        
                % === 2. Load NatNet .NET assembly ===
                dllFile = app.getNatNetDLLPath();
                if ~libisloaded('NatNetML')
                    NET.addAssembly(dllFile);
                end
                
                
                % === 3. Initialize NatNet client ===

                app.natnet_client.HostIP = app.optitrack_serverIP;
                app.natnet_client.ClientIP = app.optitrack_localIP;
                app.natnet_client.connect();
        
                success = true;
                msg = '✅ OptiTrack natnet client connected.';
            catch ME
                msg = sprintf('❌ OptiTrack init error: %s', ME.message);
                warning(msg);
            end
        end


        function StartOptiTrackRecording(app)
            if isempty(app.natnet_client)
                [success, msg] = InitOptiTrack(app);
                if ~success
                    app.Msg1.HTMLSource = sprintf('<p>%s</p>', msg);
                    return;
                end
            end
            app.optitrack_recording = true;
            app.optitrack_data = struct('time', {}, 'markers', {});
            app.optitrack_filename = sprintf('%s_optitrack.mat', app.FullPathFilename);
        
            app.optitrack_timer = timer('ExecutionMode','fixedRate',...
                'Period',1/app.optitrack_pollRate,...
                'TimerFcn',@(~,~)CollectOptiTrackFrame(app));
        
            start(app.optitrack_timer);
            app.logMessage('OptiTrack recording started.');
        end
        function CollectOptiTrackFrame(app)
            try
                frame = app.natnet_client.GetLastFrameOfData();
                t = frame.fTimestamp;
                if frame.nMarkers > 0
                    n = frame.nMarkers;
                    markerPos = zeros(n,3);
                    for i = 1:n
                        m = frame.OtherMarkers(i);
                        markerPos(i,:) = [m.x, m.y, m.z];
                    end
                    app.optitrack_data(end+1).time = t;
                    app.optitrack_data(end).markers = markerPos;
                end
            catch
                % ignore transient errors
            end
        end
        function StopOptiTrackRecording(app)
            if app.optitrack_recording
                stop(app.optitrack_timer);
                delete(app.optitrack_timer);
                app.optitrack_recording = false;
                
                if ~isempty(app.natnet_client)
                    try
                        app.natnet_client.disconnect();
                        clear app.natnet_client;
                    catch
                        app.logMessage('Failed to gracefully disconnect NatNet client.', 'warning');
                    end
                end

                
                optitrack_data = app.optitrack_data; %#ok<NASGU>
                save(app.optitrack_filename, 'optitrack_data');
                app.logMessage(['OptiTrack data saved to ', app.optitrack_filename]);

            end
        end

        function [success, msg] = sendMotiveCommand(app, cmd)
        % sendMotiveCommand  Send remote command to Motive over TCP
        % Requires Motive "Remote Control" enabled.
        % Common commands: 'StartRecording', 'StopRecording', 'SetRecordingName <name>', 'SaveTake'
        
            success = false;
            msg = '';
            try
                % Motive listens on port 1510 by default
                host = app.optitrack_serverIP;
                port = 1510;
        
                % Open TCP connection
                t = tcpclient(host, port, 'Timeout', 2);
                % Send ASCII command
                write(t, uint8([cmd newline]));
                pause(0.1); % short delay
                clear t;
        
                success = true;
                msg = sprintf('Sent command to Motive: %s', cmd);
                fprintf('%s\n', msg);
            catch ME
                msg = sprintf('Failed to send Motive command (%s): %s', cmd, ME.message);
                warning(msg);
            end
        end

        function success = SaveAudio(app)
            success = false;
            try
                if isrecording(app.audioRecObj), stop(app.audioRecObj);end
                s = getaudiodata(app.audioRecObj);
                delete(app.audioRecObj);
            catch
                delete(app.audioRecObj);
            end
            nChannels = size(s, 2);

            % Remap 4-channel input to 3-channel output:
            % Input:  Ch1=Audio, Ch2=Telemed sync, Ch3=(unused), Ch4=OptiTrack sync
            % Output: Ch1=Audio, Ch2=Telemed sync, Ch3=OptiTrack sync
            if nChannels >= 4
                s_output = [s(:,1), s(:,app.which_channel_is_telemed_sync), s(:,app.which_channel_is_optitrack_sync)];
                app.logMessage(sprintf('Remapped %d channels to 3-channel output (Audio, Telemed sync, OptiTrack sync)', nChannels), 'info');
            elseif nChannels >= 3
                % Fallback: if only 3 channels available, assume Ch3 is Telemed sync
                s_output = [s(:,1), s(:,app.which_channel_is_telemed_sync), zeros(size(s,1),1)];
                app.logMessage(sprintf('Warning: Only %d channels recorded. OptiTrack sync channel will be silent.', nChannels), 'warning');
            else
                % Fallback: save as-is if fewer than 3 channels
                s_output = s;
                app.logMessage(sprintf('Warning: Only %d channels recorded. Saving as-is.', nChannels), 'warning');
            end

            % Normalize each output channel if requested
            if app.Ck_ifNormAudio.Value
                for i = 1:size(s_output, 2)
                    s1 = s_output(:,i); s1 = s1 - mean(s1);
                    s1 = s1 ./ max(abs(s1)); s_output(:,i) = s1;
                end
            end

            try
                audiowrite(app.AudioFullPathFilename, s_output, app.audio_srate);
            catch
                success = false;  return;
            end
            success = true;
        end %SaveAudio

        function appSetStatus(app, status)
            busy = false; active = false;
            %% Disable the following functions/buttons when busy (enable them when ready):
            funcsToBeDisabledIfBusy = ...
                {app.Bn_Record; app.Ed_FilePrefix; app.Mn_Setting; ...
                 app.Bn_SelectOutputFolder; app.Ed_RecordDur; app.Ck_RecordByDur; ...
                 app.Pn_Filename; app.Mn_Setting; app.Mn_Info; ...
                 app.Ck_ifNormAudio; app.Ck_ifRecordTelemed; app.Mn_Tools; ...
                 app.Ck_AppendDate; app.Ck_AppendTime
                };
            %%
            funcsToBeDisabledIfInactive = ...
                {app.Bn_Record; app.Bn_Stop; app.Mn_Tools; app.Mn_Info; ...
                 app.Ed_RecordDur; app.Ck_RecordByDur;  app.Ed_FilePrefix;  ...
                 app.Ck_AppendDate; app.Ck_AppendTime ...
                };
            %%
            if strcmpi(status, 'busy')
                active = true; busy = true; app.status = 'busy';               
            elseif strcmpi(status, 'ready')
                active = true; busy = false; app.status = 'ready';
                app.logMessage('Ready.', 'info');
            elseif strcmpi(status, 'inactive')  
                active = false; busy = false; app.status = 'inactive';  
            end
            if busy
                app.flag_busy = true;
                for i = 1:numel(funcsToBeDisabledIfBusy), funcsToBeDisabledIfBusy{i}.Enable = 'off'; end
            else 
                app.flag_busy = false;
                for i = 1:numel(funcsToBeDisabledIfBusy), funcsToBeDisabledIfBusy{i}.Enable = 'on';end
                app.Bn_Record.Value = 0;
                updateFilenameTemplate(app);
            end 
            if ~active
                for i = 1:numel(funcsToBeDisabledIfInactive), funcsToBeDisabledIfInactive{i}.Enable = 'off'; end
            end 
        end %appSetStatus
        function yes = isActive(app)
          if strcmpi(app.status, 'inactive'), yes = false; else, yes = true; end
        end % isActive
        function yes = isBusy(app)
          if strcmpi(app.status, 'busy'), yes = true; app.flag_busy = true; else, yes = false; app.flag_busy = false; end
        end % isActive

        function GenFilename(app)
            dateStr = string(app.recordStartTime, app.dateFormat);
            timeStr = string(app.recordStartTime, app.timeFormat);
            fn = app.save_file_prefix;
            if app.Ck_AppendDate.Value, fn = sprintf('%s_%s', strip(fn, 'right', '_'), dateStr); end 
            if app.Ck_AppendTime.Value, fn = sprintf('%s_%s', strip(fn, 'right', '_'), timeStr); end 
            app.Ed_Filename.Value = fn;
            app.FullPathFilename = fullfile(app.OutputPath, app.Ed_Filename.Value); 
            app.AudioFullPathFilename = sprintf('%s%s', app.FullPathFilename, app.audioExt); 
            app.TvdFullPathFilename = sprintf('%s%s', app.FullPathFilename, app.tvdExt); 
            app.AudioFullPathFilename = app.NextAvailableFileName(app.AudioFullPathFilename);
            app.TvdFullPathFilename = app.NextAvailableFileName(app.TvdFullPathFilename);
            [~,fn,~]= fileparts(app.AudioFullPathFilename);
            app.Ed_Filename.Value = fn;
        end%GenFilename
        %% 
        function updateFilenameTemplate(app)
            prefix = app.Ed_FilePrefix.Value;
            exp = '[a-zA-Z_0-9-]+';
            prefix = regexp(prefix, exp, 'match');
            prefix = strjoin(prefix,'');
            app.save_file_prefix = strip(prefix, 'right', '_');
            if isempty(app.save_file_prefix), app.save_file_prefix= '_'; end
            fn = app.save_file_prefix;
            if app.Ck_AppendDate.Value, fn = sprintf('%s_%s', strip(fn, 'right', '_'), app.dateFormat); end 
            if app.Ck_AppendTime.Value, fn = sprintf('%s_%s', strip(fn, 'right', '_'), app.timeFormat); end 
            app.Ed_Filename.Value = fn;          
        end %updateFilenameTemplate

        %% another log function for the app
        function logMessage(app, msg, type)
            if nargin < 3 || isempty(type)
                type = 'info';
            end
        
            t = datestr(now,'HH:MM:SS');
        
            % Colors
            switch lower(type)
                case 'error'
                    color = '#ff4444';
                case 'warning'
                    color = '#ff9900';
                otherwise
                    color = '#000000'; % info = black
            end
        
            % Format one line
            msg = strrep(msg, newline, '<br>');
            newLine = sprintf('<p style="color:%s; margin:0;">[%s] %s</p>', color, t, msg);
        
            % Append to stored log HTML
            if isfield(app.Msg1.UserData, 'logHTML')
                app.Msg1.UserData.logHTML = [app.Msg1.UserData.logHTML newLine];
            else
                app.Msg1.UserData.logHTML = newLine;
            end
        
            % Rebuild only inner content
            fullHTML = [
                "<html><head><style>", ...
                "body {font-family: Consolas; font-size:8px; line-height:1.0; margin:2px;}", ...
                "#logbox {height:100%; overflow-y:auto; display:flex; flex-direction:column-reverse;}", ...
                "#content {display:flex; flex-direction:column;}", ...
                "</style></head><body>", ...
                "<div id='logbox'><div id='content'>", ...
                app.Msg1.UserData.logHTML, ...
                "</div></div></body></html>"
            ];
        
            app.Msg1.HTMLSource = strjoin(fullHTML, "");
            drawnow limitrate nocallbacks; % ensure refresh
        
            % Console echo
            fprintf('[%s] %s\n', t, msg);
        end

        function [success, msg] = DetectEchoWave(app)
            success = false; msg = '';
            if ~ispc, msg = 'Error in EchoWave: Not PC!';  return; end
            if ~app.isAdminPC, msg  = 'Error in EchoWave: Not Admin!';  return; end
            if isempty(app.EchoWavePath)
                app.EchoWavePath =  app.FindTelemedEchoWavePath; 
                if isempty(app.EchoWavePath)
                    msg = 'Error in EchoWave: path not found!';  return;

        function [sig, sr, locs, frameStart] = TrimTelemedAudio(FName, resampleRate, outFName, fps, sync_channel)
        % Trim the begining of Telemed audio using the sync-pulse signals.
        % Now works with 3-channel audio: Ch1=Audio, Ch2=Telemed sync, Ch3=OptiTrack sync
        % WR Chen 19DEC2022 | Updated for 3-channel format 2025
        if nargin < 2, resampleRate = [];end
        if nargin < 3, outFName = []; end
        if nargin < 4, fps = []; end
        if nargin < 5 || isempty(sync_channel), sync_channel = 2; end  % Default: Ch2 = Telemed sync
        if isempty(fps), max_fps = 300; else, max_fps = fps * 1.3; end
        [locs, audio, syncs, sr] = ParseTelemedSyncPulse(FName, sync_channel, max_fps);
        nSamplesAudio = size(audio,1);
        frameStart = mean(locs(1:2)); % Get the midpoint between the time of starting scanning and the completion of scanning first frame.
        frameDur = median(diff(locs(2:end)));
        frameEnd = locs(end) + frameDur;
        frameStartN = floor(frameStart * sr)+1;
        frameEndN = floor(frameEnd * sr)+1;
        if frameEndN > nSamplesAudio, frameEndN = nSamplesAudio; end
        sig = audio(frameStartN:frameEndN);
        if ~isempty(resampleRate),  sig = resample(sig, resampleRate,sr); sr = resampleRate; end
        if ~isempty(outFName), audiowrite(outFName, sig, sr); end
            function [locs, audio, syncs, sr] = ParseTelemedSyncPulse(FName, sync_channel, max_fps)
                % Parse sync pulses from 3-channel WAV file
                % Ch1=Audio, Ch2=Telemed sync, Ch3=OptiTrack sync
                if nargin < 2, sync_channel = 2; end  % Default: Ch2 = Telemed sync
                if nargin < 3 || isempty(max_fps), max_fps = 300; end
                audio_channel = 1;  % Audio is always channel 1 in the 3-channel output
                [s, sr]= audioread(FName); nChannels = size(s,2);
                for i = 1:nChannels, s1 = s(:,i); s(:,i) = s1 ./ max(abs(s1)); end
                audio = s(:,audio_channel); syncs = s(:,sync_channel); 
                nSamples = length(audio); dur = (nSamples-1) / sr; 
                syncs = syncs ./ max(abs(syncs));   min_fps = 20; 
                [pks, locs] = findpeaks(syncs,sr, 'MinPeakHeight', 0.3, 'MinPeakDistance',1/max_fps);
                intervals = diff(locs);
            end % ParseTelemedSyncPulse
        end %TrimTelemedAudio

        function [status,cmdout] = ffmpegCombineVideoAudio(input_video_filename, input_audio_filename, output_video_filename, opt)
            % Combine video and audio using FFMPEG.
            % REQUIRE: FFMPEG properly installed and searchable in the system PATH. 
            % WR Chen  23MAY2023
            % Latest:  15JUN2023 : added pixel format
            arguments
                input_video_filename char {mustBeText}
                input_audio_filename char {mustBeText}
                output_video_filename char {mustBeText}
                opt.vcodec char {mustBeText} = '-c:v libx264'
                opt.acodec char {mustBeText} = '-c:a copy'
                opt.preset char {mustBeText} = '-preset slow'
                opt.crf char {mustBeText} = '-crf 23' % crf: 0:best ~ 51:worst; Default=23; 17~18: visually lossless
                opt.pixel_format char {mustBeText}  = '-pix_fmt yuv420p' 
                opt.ifUseShortest logical = false % if true, the video and audio will be trimmed to whichever is the shortest.
                opt.ffmpegPath char {mustBeText} = ''
            end
            if ~exist(input_video_filename, 'file'), fprintf('Cannot find: %s\n', input_video_filename); return; end
            if ~exist(input_audio_filename, 'file'), fprintf('Cannot find: %s\n', input_audio_filename); return; end
            if opt.ifUseShortest, useShortest = '-shortest'; else, useShortest = ''; end
            % check FFMPEG:
            if ~isempty(opt.ffmpegPath)
                ffmpegExe = ['"', fullfile(opt.ffmpegPath, 'ffmpeg'), '"'];
            else
                ffmpegExe = 'ffmpeg';
            end
            %%   
            aFn = input_audio_filename; vFn = input_video_filename;
            outFn = output_video_filename;
            command = sprintf(...
             '%s -i "%s" -i "%s" -map 0:v -map 1:a %s %s %s %s %s -strict -2 %s -y "%s"', ...
                ffmpegExe, vFn, aFn, opt.vcodec, opt.pixel_format, opt.preset, opt.crf, opt.acodec, useShortest, outFn);
            % tic
            [status,cmdout] = system(command,'-echo');
            % toc    
        end % ffmpegCombineVideoAudio

            %             app.useAudioDeviceReader = true;
            %             app.audioData = [];
            %             app.audioTimer = timer( ...
            %                 'ExecutionMode','fixedRate', ...
            %                 'Period',0.05, ...
            %                 'TimerFcn',@(~,~)captureAudioFrame(app));
            %             start(app.audioTimer);
            % 
            %             app.logMessage(sprintf('🎧 Recording via WASAPI (%d ch).', app.num_audio_channels), 'info');
            %         catch ME
            %             % Failed: revert to audiorecorder
            %             app.logMessage(sprintf('⚠️ WASAPI failed: %s → using audiorecorder fallback.', ME.message), 'warning');
            %         end
            %     end
            % 
            %     % === Fallback if WASAPI not available ===
            %     if ~app.useAudioDeviceReader
            %         app.num_audio_channels = 2; % only stereo supported
            %         app.audioRecObj = audiorecorder(app.audio_srate, app.audio_bit_rate, app.num_audio_channels, app.audio_dev_idx);
            %         record(app.audioRecObj);
            %         pause(app.timerPeriod);
            %         app.logMessage('🎙️ Recording via audiorecorder (2-ch MME driver).', 'info');
            %     end
            % 
            % catch ME
            %     app.logMessage(sprintf('❌ Audio initialization error: %s', ME.message), 'error');
            %     app.isRecording = false;
            %     app.Bn_Stop.Enable = 'off';
            %     app.Bn_Record.Enable = 'on';
            %     appSetStatus(app, 'ready');
            %     return;
            % end

            %% Telemed Record:
            if app.Ck_ifRecordTelemed.Value
                EchoWaveFreeze(app);
                EchoWaveStartRecord(app); 
            end
            %% OptiTrack Record (hardware-triggered)
            if app.Ck_ifRecordOptitrack.Value
                if ~isMotiveRunning(app)
                    app.Msg1.HTMLSource = ...
                        '<p style="color:red;">⚠️ Motive is not running!<br>Start OptiTrack Motive before recording.</p>';
                    app.Ck_ifRecordOptitrack.Value = false;
                    figure(app.UIFigure);
                    return;
                end
            
                [success, msg] = InitOptiTrack(app);
                if ~success
                    % app.Msg1.HTMLSource = sprintf('<p style="color:red;">%s</p>', msg);
                    app.logMessage(sprintf('%s', msg), 'error')
                    return;
                end
            
                % Set take name and folder (Motive will save .tak automatically)
                takeDir  = 'C:\Users\Boram Kim\Documents\OptiTrack\Default';
                takeName = app.Ed_Filename.Value;
                app.natnet_client.setTakeName(app.Ed_Filename.Value);
            
                app.motiveTrigger.start = datetime('now');
                app.logMessage('🔴 Motive: startRecord()');
                app.natnet_client.startRecord();
            
                % (optional) also start MATLAB marker polling
                StartOptiTrackRecording(app);
            end



            %% Recording timer:
            dur = 0; audioSignalLength = 0; previousAudioSignalLength = 0;
            Msg1HTML = 'Ready.';
            while app.Bn_Record.Value
                currentTime = datetime('now'); durObj = currentTime - app.recordStartTime;              
                dur = round(seconds(durObj), 3); 
                app.Ed_TimeRecorded.Value = string(durObj, 'hh:mm:ss.SSS');
                try
                    pause(app.timerPeriod);
                    S = getaudiodata(app.audioRecObj); s = S(end,:); 
                    previousAudioSignalLength = audioSignalLength;
                    audioSignalLength = size(S,1);  
                    audioSignalLengthDiff = audioSignalLength - previousAudioSignalLength;
                catch
                    app.isRecording= false; audioRecordedSuccess = false; break 
                end
                if audioSignalLengthDiff < 1, app.isRecording= false; audioRecordedSuccess = false; Msg1HTML = ''; break; end
                if s(1) ~= 0, app.Lp_AudSig1.Color = app.hasSignalColor; end
                if size(s, 2) > 1 && s(2) ~= 0, app.Lp_AudSig2.Color = app.hasSignalColor; end
                if size(s, 2) > 2 && s(3) ~= 0, app.Lp_AudSig3.Color = app.hasSignalColor; end
                if size(s, 2) > 3 && s(4) ~= 0, app.Lp_AudSig4.Color = app.hasSignalColor; end
                if dur > app.maxTimeOut
                    % app.Msg1.HTMLSource = '<p>Max time out!</p>';
                    app.logMessage('Max time out!', 'warning')
                    app.isRecording= false; break
                end
                if app.Ck_RecordByDur.Value && dur > app.Ed_RecordDur.Value
                    app.isRecording= false; break
                end
            end % while app.isRecording
            app.Bn_Stop.Enable = 'off'; 

            app.Lp_AudSig1.Color = app.noSignalColor; app.Lp_AudSig2.Color = app.noSignalColor;
            % Stop recording:
            if app.Ck_ifRecordTelemed.Value

            end
            try
                stop(app.audioRecObj); 
            catch
                audioRecordedSuccess = false; 
            end
            app.recordStopTime = datetime('now');   
            % Save data:
            if app.Ck_ifRecordTelemed.Value, EchoWaveStopAndSave(app); end
            if app.Ck_ifRecordOptitrack.Value
                app.motiveTrigger.stop = datetime('now');
                app.logMessage('🟢 Motive: stopRecord()');
                try
                    app.natnet_client.stopRecord();  % stop & save .tak
                catch ME
                    app.logMessage(sprintf('Failed to stop Motive recording: %s', ME.message), 'error');
                end
                StopOptiTrackRecording(app);
            end


            if audioRecordedSuccess
                success = SaveAudio(app); 
                if success
                     if app.Ck_ifRecordTelemed.Value
                         msg = 'success-US-audio';
                     else
                         msg = 'success-audio';
                     end
                else
                    msg = 'write-audio-failed';     
                    Msg1HTML = '<p>Fail to write audio file! <br>Permission denied. <br> Try another output folder.</p>';
                end 
            else 
                Msg1HTML = '<p>Error in audio recording!'; 
                msg = 'audio-error';
            end
            % do_log(app, msg);  
            app.logMessage(msg)
            %%       
            appSetStatus(app, 'ready');
            % app.Msg1.HTMLSource = Msg1HTML;
            app.logMessage(Msg1HTML)
            %% test
            % app.asmcmd.GoToFrame1n(nFrames, true);
            % img = uint8(app.asmcmd.GetLoadedFrameGray());
        end

        % Menu selected function: Mn_AudioSettings
        function Mn_AudioSettingsMenuSelected(app, event)

            defaultAnswer = {sprintf('%d', app.audio_srate), sprintf('%d', app.num_audio_channels)};
            questions = {'Audio sampling rate (Hz)', 'Number of audio channels'};
            answers = inputdlg(questions,...
             '',1, defaultAnswer);
            figure(app.UIFigure); % Get focus back to main figure
            if isempty(answers), return; end % user cancelled
            app.Msg1.HTMLSource = 'Ready.';
            %% Samping rate:
            ans1 = round(str2double(answers{1}));
            if ~isnan(ans1) && ans1 >= 16000 && ans1 <= 96000
                app.audio_srate = ans1;
                app.logMessage(sprintf('Set sampling rate to %d Hz.', ans1), 'info');
            else
                app.logMessage('Unsupported sampling rate.', 'warning'); 
            end %
            %% Number of audio channels:
            current_num_audio_channels = app.num_audio_channels; % store the current state of num_audio_channels.
            ans2 = round(str2double(answers{2}));
            if isnan(ans2) || ans2 < 1
                app.Msg1.HTMLSource = ...
                   sprintf('%s <br> Unsupported number of channels.', app.Msg1.HTMLSource);
                return
            end
            app.num_audio_channels = ans2;
            [support, msg] = CheckIfAudioDeviceSettingSupported(app);
            if ~support 
                app.num_audio_channels = current_num_audio_channels;  % change back to the previous state.
                UpdateRefreshAudioStatus(app);
                app.Msg1.HTMLSource = msg; 
                return; 
            end 
            app.Msg1.HTMLSource = ...
               sprintf('%s <br> Set number of audio channgels to %d.', app.Msg1.HTMLSource, ans2);
            appSetStatus(app, 'ready');  
        end

        % Menu selected function: Mn_ConvertTVDtoMP4viaAVI
        function Mn_ConvertTVDtoVideoMenuSelected(app, event)
            if app.flag_busy, return; end
            previous_status = app.status; appSetStatus(app, 'busy'); 
            [success, msg] = DetectEchoWave(app);   
            toFormat = event.Source.UserData.toFormat; % mp4 or avi_comp
            videoExt = strrep(toFormat, '_comp','');
            if ~success, uialert(app.UIFigure,msg,'EchoWave error'); appSetStatus(app, previous_status); return;end
           %% Check ffmpeg:
            app.ffmpegFound = app.hasFFmpegInstalled();
            if ~app.ffmpegFound
                msg = {'FFMPEG not found! ', 'TVD will be converted but NOT combined with audio.', 'Synchronized audio will be saved as .flac file.', ' '};
            else
                msg = {''};
            end
            mask1 = sprintf('%s%s*%s', app.OutputPath, filesep, app.tvdExt);            
            % mask2 = sprintf('%s%s*%s', app.OutputPath, filesep, app.audioExt);
            fl = app.mygfl(mask1,0); % fl2 = app.mygfl(mask2,0);
            % ix = ismember(fl, fl2); 
            % if ~any(ix), app.Msg1.HTMLSource = 'No TVD files!'; appSetStatus(app, previous_status); return;end
            if isempty(fl), app.Msg1.HTMLSource = 'No TVD files!'; appSetStatus(app, previous_status); return;end    
            nfl = numel(fl);
            %% Confirm:
            msg = [msg, {sprintf('%d TVD files found.', nfl)}];
            msg = [msg, {'Video conversion takes very long time.', ...
                'Remember to max out Echowave resolution', 'and set MP4 bit rate > 8000 Kbps.', 'Continue?'}];
            selection = uiconfirm(app.UIFigure, msg,'Confirm conversion', 'Icon','warning');
            switch selection, case 'OK',  case 'Cancel',  appSetStatus(app, previous_status); return; end   
            %% Create temp folder:
            tmpPath = fullfile(app.OutputPath, app.tmpFolder);
            [success, ~]=mkdir(tmpPath);
            if ~success, uialert(app.UIFigure,'Failed to create temp folder. Access denied!','Error');  appSetStatus(app, previous_status); return;end

            h_prog = uiprogressdlg(app.UIFigure,'Title','Converting TVD to MP4','Cancelable','on');
            %%
            for ifl = 1:nfl
                % Check for Cancel button press
                if h_prog.CancelRequested,  break;  end
                fn = fl{ifl}; inFn = fullfile(app.OutputPath, fn); toFn =  fullfile(tmpPath, fn); 
                % Update progress, report current estimate
                h_prog.Value = ifl/nfl;  h_prog.Message = sprintf('%d of %d file(s): %s', ifl, nfl, fn);
                %
                tvdFN = [inFn, app.tvdExt]; audioFN = [inFn, app.audioExt];
                tmpTvdFN = fullfile(tmpPath, [fn app.tvdExt]); tmpVideoFN = fullfile(tmpPath, [fn '.' videoExt]); 
                outVideoFN = fullfile(app.OutputPath, [fn '.' 'mp4']);
                [status,msg] = copyfile(tvdFN, tmpTvdFN); 
                info = EchoWaveGetTVDinfo(app,tmpTvdFN);
                if ~isempty(info) && isfield(info, 'fps') && info.fps > 0, fps = info.fps; else, fps = [];  end
                ConvertTVDtoVideo(app, tmpPath, toFormat);

                if exist(audioFN, 'file')
                    % Note: In the 3-channel output WAV, Telemed sync is channel 2 (not hardware channel 3)
                    sync_channel = 2;  % Ch2 = Telemed sync in the saved 3-channel WAV file
                    trimmedAudioFN = fullfile(app.OutputPath, [fn '.flac']);
                    [sig, sr, locs, frameStart] = app.TrimTelemedAudio(audioFN, [], trimmedAudioFN, fps, sync_channel);
                    
                    if app.ffmpegFound
                        [status,cmdout] = app.ffmpegCombineVideoAudio(tmpVideoFN, trimmedAudioFN, outVideoFN);
                    else
                        [status,msg] = copyfile(tmpVideoFN, outVideoFN); 
                    end
                else
                    % No audio. TVD only
                    if app.ffmpegFound
                         [status,cmdout] = app.ffmpegConvertOneFile_simple(tmpVideoFN, outVideoFN);
                     else
                        [status,msg] = copyfile(tmpVideoFN, outVideoFN); 
                    end
                end
                delete(tmpTvdFN);   delete(tmpVideoFN); 
            end   
            %% Delete temp folder:
             [success, msg] = app.deleteFolder(tmpPath);           
            appSetStatus(app, previous_status);
            close(h_prog);
        end

    % App creation and deletion
    methods (Access = public)

        % Construct app
        function app = MewRecorder

            runningApp = getRunningApp(app);

            % Check for running singleton app
            if isempty(runningApp)

                % Create UIFigure and components
                createComponents(app)

                % Register the app with App Designer
                registerApp(app, app.UIFigure)

                % Execute the startup function
                runStartupFcn(app, @startupFcn)
            else

                % Focus the running singleton app
                figure(runningApp.UIFigure)

                app = runningApp;
            end

            if nargout == 0
                clear app
            end
        end

        % Code that executes before app deletion
        function delete(app)

            % Delete UIFigure when app is deleted
            delete(app.UIFigure)
        end
    end
